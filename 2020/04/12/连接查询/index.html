SELECT department_name,COUNT’‘ 员工个数
FROM departments d，employees e
WHERE e.’department_id’=d.’department_id’
GROUP BY department_name
HAVING COUNT()>3;
ORDER BY COUNT(*) DESC;

SELECT department_name,COUNT’‘ 员工个数
FROM employees e
INNER JOIN departments d
ON e.’department_id’=d.’department_id’
GROUP BY department_name
HAVING COUNT()>3;
ORDER BY COUNT(*) DESC;
